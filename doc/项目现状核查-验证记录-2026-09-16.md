实际执行的主要 SQL：

```sql
SELECT 'keys',count(*),
 count(*) FILTER(WHERE max_budget IS NOT NULL),
 count(*) FILTER(WHERE rpm_limit IS NOT NULL),
 count(*) FILTER(WHERE tpm_limit IS NOT NULL),
 count(*) FILTER(WHERE max_parallel_requests IS NOT NULL),
 count(*) FILTER(WHERE team_id IS NOT NULL),
 count(*) FILTER(WHERE expires IS NOT NULL),
 count(*) FILTER(WHERE blocked IS TRUE)
FROM "LiteLLM_VerificationToken";

SELECT 'users',count(*),
 count(*) FILTER(WHERE sso_user_id IS NOT NULL),
 count(*) FILTER(WHERE max_budget IS NOT NULL)
FROM "LiteLLM_UserTable";

SELECT 'teams',count(*),count(*) FILTER(WHERE max_budget IS NOT NULL)
FROM "LiteLLM_TeamTable";

SELECT 'spend',count(*),
 count(*) FILTER(WHERE total_tokens>0),
 count(*) FILTER(WHERE spend>0),
 count(*) FILTER(WHERE messages IS NOT NULL AND messages::text NOT IN ('null','[]','{}','""')),
 count(*) FILTER(WHERE response IS NOT NULL AND response::text NOT IN ('null','[]','{}','""')),
 min("startTime"),max("startTime")
FROM "LiteLLM_SpendLogs"; SELECT count(*), count(*) FILTER(WHERE spend<>0), min(spend), max(spend) FROM "LiteLLM_SpendLogs";
```
